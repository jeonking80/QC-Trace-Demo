```bash
dotnet test                                   # 규칙 81 + 통합 156 (Docker 필요, Testcontainers)
docker compose up -d                          # 앱 기동

dotnet run --project tools/Reconcile          # 독립 대조 (불일치 있으면 종료 코드 1)
```
